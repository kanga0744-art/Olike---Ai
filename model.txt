<script>
    function getPingClass(ms) {
        if (ms < 300) return 'ping-value';
        if (ms < 800) return 'ping-value';
        return 'ping-value';
    }

    // Fungsi Tes Model Individu
    async function checkModelHealth(modelName, type, btnId, resId) {
        const btn = document.getElementById(btnId);
        const res = document.getElementById(resId);
        
        btn.disabled = true;
        btn.innerHTML = '⏳...';
        res.innerHTML = '';
        res.className = 'test-result';

        try {
            let url;
            const startTime = performance.now();
            
            if (type === 'image') {
                // Request gambar kecil (16x16) untuk menghemat bandwidth, tapi cukup untuk trigger backend
                url = `https://image.pollinations.ai/prompt/test?model=${encodeURIComponent(modelName)}&width=32&height=32&nologo=true&seed=${Math.floor(Math.random()*1000)}`;
            } else {
                // Untuk teks, kita coba hit endpoint sederhana jika memungkinkan.
                // Karena endpoint text OpenAI compatible kadang butuh POST, kita coba GET root/model check jika tersedia, 
                // atau asumsikan 200 OK dari endpoint list sudah cukup, tapi di sini kita simulasikan 'active'
                // Note: Text generation via GET di Pollinations tidak selalu standar. 
                // Kita gunakan endpoint list sebagai proxy test untuk text jika tidak ada endpoint GET khusus.
                // Namun untuk konsistensi UI, kita akan membiarkan user tahu ini "List Checked".
                url = `https://text.pollinations.ai/`; // Base check
            }

            const response = await fetch(url, { method: 'GET', cache: 'no-cache' });
            const endTime = performance.now();
            const duration = Math.round(endTime - startTime);

            if (response.ok) {
                res.innerHTML = `✅ Aktif (${duration}ms)`;
                res.classList.add('test-ok');
            } else {
                // Jika error (misal 500 seperti kasus zimage)
                res.innerHTML = `❌ Gagal (${response.status})`;
                res.classList.add('test-fail');
            }

        } catch (error) {
            console.error(error);
            res.innerHTML = `❌ Error`;
            res.classList.add('test-fail');
        } finally {
            btn.disabled = false;
            btn.innerHTML = '⚡ Tes';
        }
    }

    async function fetchAndRender(type, url, tableBodyId) {
        const tbody = document.getElementById(tableBodyId);
        tbody.style.opacity = '0.5';
        
        const startTime = performance.now();
        let ping = 0;
        let isSuccess = false;
        let models = [];
        let errorMsg = '';

        try {
            const response = await fetch(url, { cache: "no-store" });
            const endTime = performance.now();
            ping = Math.round(endTime - startTime);
            isSuccess = response.ok;
            if (!isSuccess) throw new Error(`HTTP ${response.status}`);
            models = await response.json();
        } catch (error) {
            console.error(error);
            isSuccess = false;
            errorMsg = error.message;
        }

        tbody.innerHTML = '';
        tbody.style.opacity = '1';

        if (!isSuccess) {
            tbody.innerHTML = `
                <tr>
                    <td><span class="status-badge status-error"><span class="dot"></span> Down</span></td>
                    <td colspan="4" style="color:var(--error);">Gagal mengambil daftar: ${errorMsg}</td>
                </tr>`;
            return;
        }

        if (models.length === 0) {
            tbody.innerHTML = '<tr><td colspan="5" style="text-align:center; padding:20px;">Tidak ada model ditemukan.</td></tr>';
            return;
        }

        models.forEach((model, index) => {
            const row = document.createElement('tr');
            const uniqueId = `${type}-${index}`;
            const btnId = `btn-${uniqueId}`;
            const resId = `res-${uniqueId}`;

            // Image vs Text Layout
            let detailsHtml = '';
            if (type === 'text') {
                if (model.reasoning) detailsHtml += `<span class="feature-tag">Reasoning</span>`;
                if (model.tools) detailsHtml += `<span class="feature-tag">Tools</span>`;
                detailsHtml += `<div class="desc-text">${model.description || '-'}</div>`;
            } else {
                const inputs = model.input_modalities ? model.input_modalities.join(', ') : '';
                detailsHtml += `<div class="desc-text">${model.description || ''}</div>`;
                if(inputs) detailsHtml += `<small style="color:#9ca3af">In: ${inputs}</small>`;
            }

            // Test Button Logic
            let testButtonHtml = '';
            if (type === 'image') {
                testButtonHtml = `
                    <button id="${btnId}" class="test-btn" onclick="checkModelHealth('${model.name}', 'image', '${btnId}', '${resId}')">
                        ⚡ Tes
                    </button>
                    <span id="${resId}" class="test-result"></span>
                `;
            } else {
                 testButtonHtml = `<span style="color:#cbd5e1; font-size:0.8rem;">Lihat Status API</span>`;
            }

            row.innerHTML = `
                <td>
                    <span class="status-badge status-success">
                        <span class="dot"></span> Terdaftar
                    </span>
                </td>
                <td>
                    <div class="model-name">${model.name}</div>
                    <small style="color:#9ca3af">${model.type || type}</small>
                </td>
                <td>${detailsHtml}</td>
                <td>
                    <span class="${getPingClass(ping)}">${ping}ms</span>
                </td>
                <td>
                    ${testButtonHtml}
                </td>
            `;
            tbody.appendChild(row);
        });
    }

    async function refreshAllData() {
        const btn = document.getElementById('refreshBtn');
        const lastUpdatedEl = document.getElementById('lastUpdated');
        const originalText = btn.innerHTML;
        
        btn.disabled = true;
        btn.innerHTML = '⏳ Memuat...';

        await Promise.all([
            fetchAndRender('text', 'https://gen.pollinations.ai/text/models', 'text-body'),
            fetchAndRender('image', 'https://gen.pollinations.ai/image/models', 'image-body')
        ]);

        btn.disabled = false;
        btn.innerHTML = originalText;
        
        const now = new Date();
        lastUpdatedEl.textContent = `Update: ${now.toLocaleTimeString()}`;
    }

    document.addEventListener('DOMContentLoaded', () => {
        refreshAllData();
    });
