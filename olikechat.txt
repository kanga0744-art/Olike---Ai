            document.getElementById('btn-ratio-landscape').className = `ratio-btn ${ratio === 'landscape' ? 'active' : ''}`;
        }

        // --- CHECK POLLEN BALANCE ---
        async function checkPollenBalance() {
            if(!apiKey) {
                pollenStatusDiv.style.display = 'none';
                return;
            }

            const baseUrl = 'https://gen.pollinations.ai/account';
            const headers = {
                'Authorization': `Bearer ${apiKey}`,
                'Content-Type': 'application/json'
            };

            try {
                // 1. Ambil Data Profil
                const profileReq = await fetch(`${baseUrl}/profile`, { headers });
                if (!profileReq.ok) throw new Error(`Profile Error: ${profileReq.status}`);
                const profile = await profileReq.json();

                // 2. Ambil Data Saldo
                const balanceReq = await fetch(`${baseUrl}/balance`, { headers });
                if (!balanceReq.ok) throw new Error(`Balance Error: ${balanceReq.status}`);
                const balanceData = await balanceReq.json();
                
                // Update DOM
                document.getElementById('tierName').innerText = (profile.tier ? profile.tier : 'FREE').toUpperCase();
                document.getElementById('pollenTotal').innerText = (balanceData.balance || 0).toFixed(2);
                
                if (profile.nextResetAt) {
                    const date = new Date(profile.nextResetAt);
                    document.getElementById('resetTime').innerText = date.toLocaleString();
                } else {
                    document.getElementById('resetTime').innerText = 'Tidak tersedia';
                }
                
                pollenStatusDiv.style.display = 'block';
                updateKeyIcon(true);

            } catch (error) {
                console.error("Gagal cek saldo:", error);
                pollenStatusDiv.style.display = 'none';
                // Visual feedback invalid key
                keyIcon.innerHTML = '<i class="fas fa-exclamation-circle text-red-500"></i>';
            }
        }

        function updateModelStatusUI() {
            if (apiKey) {
                modelStatusIcon.className = "fas fa-robot text-green-glow transition-colors duration-300";
            } else {
                modelStatusIcon.className = "fas fa-robot text-zinc-600 transition-colors duration-300";

        }

        function handleKey(e) {
            if(e.key === 'Enter' && !e.shiftKey) {
                e.preventDefault();
                sendMessage();
            }
        }

        function scrollToBottom() {
            scrollArea.scrollTo({ top: scrollArea.scrollHeight, behavior: 'smooth' });
        }

        function toggleSettings() {
            document.getElementById('settingsModal').classList.toggle('active');
        }

        function saveSettings(silent = false) {
            const val = apiKeyField.value.trim();
            if(val) {
                apiKey = val;
                localStorage.setItem('pollinations_key', val);
                updateKeyIcon(true);
                checkPollenBalance();
            } else {
                apiKey = '';
                localStorage.removeItem('pollinations_key');
                updateKeyIcon(false);
                pollenStatusDiv.style.display = 'none';
            }
            
            updateModelStatusUI();
            if(!silent) toggleSettings();
        }

        function updateKeyIcon(active) {
            // Error icon handled by checkPollenBalance logic

                keyIcon.innerHTML = '<i class="fas fa-check-circle text-green-500"></i>';
            } else if (!active) {
                keyIcon.innerHTML = '<i class="fas fa-key"></i>';
            }
        }

        function resetChat() {

        // --- MAIN CHAT LOGIC ---
        async function sendMessage() {
            const text = userInput.value.trim();
            if(!text) return;

            // UI State
            userInput.value = '';
            emptyState.style.display = 'none';
            sendBtn.disabled = true;
            sendBtn.style.opacity = '0.5';

            // Append User Msg
            appendMessage('user', text);
            
            // Only add to history if it's NOT direct image mode
            // For image modes, we don't necessarily need chat history context
            if (!['zimage'].includes(model)) {
                chatHistory.push({ role: "user", content: text });
            }

            // Loading
            loader.style.display = 'block';
            scrollToBottom();

            try {
                // --- IMAGE GENERATION BRANCH ---
                if (['zimage'].includes(model)) {
                    // Logic: Auto upscale 2K based on Aspect Ratio
                    const width = aspectRatio === 'portrait' ? 1080 : 1920;
                    const height = aspectRatio === 'portrait' ? 1920 : 1080;
                    const seed = Math.floor(Math.random() * 1000000000);
                    
                    // Construct URL
                    const encodedPrompt = encodeURIComponent(text);
                    const imageUrl = `https://image.pollinations.ai/prompt/${encodedPrompt}?width=${width}&height=${height}&model=${model}&seed=${seed}&nologo=true`;
                    
                    // Create Markdown response
                    const markdownResponse = `![Generated Image](${imageUrl})`;
                    
                    // Simulate delay slightly for UX
                    await new Promise(r => setTimeout(r, 1000));
                    
                    appendMessage('assistant', markdownResponse);
                    
                } 
                // --- TEXT GENERATION BRANCH ---
                else {
                    const headers = { 'Content-Type': 'application/json' };
                    if(apiKey) headers['Authorization'] = `Bearer ${apiKey}`;

                    // FIX: Use standard chat endpoint for better reliability and JSON response
                    const res = await fetch('https://gen.pollinations.ai/v1/chat/completions', {
                        method: 'POST',
                        headers: headers,
                        body: JSON.stringify({
                            messages: chatHistory,
                            model: model,
                            // jsonMode: false // Not strictly needed for gen.pollinations.ai usually
                        })
                    });

                    if(!res.ok) throw new Error("API Error");
                    
                    const data = await res.json();
                    const reply = data.choices[0].message.content;
                    
                    chatHistory.push({ role: "assistant", content: reply });
                    appendMessage('assistant', reply);
                }

            } catch(e) {
                console.error(e);
                appendMessage('assistant', "**Error:** Gagal memproses permintaan. Cek koneksi atau API Key.");
            } finally {
                loader.style.display = 'none';
                sendBtn.disabled = false;
                sendBtn.style.opacity = '1';
                scrollToBottom();
                userInput.focus();
            }
        }

        function appendMessage(role, text) {
            const row = document.createElement('div');
            row.className = 'msg-container';
            row.style.flexDirection = role === 'user' ? 'row-reverse' : 'row';

            // Avatar
            const avatar = document.createElement('div');
            avatar.style.width = '2rem';
            avatar.style.height = '2rem';
            avatar.style.borderRadius = '50%';
            avatar.style.flexShrink = '0';
            avatar.style.display = 'flex';
            avatar.style.alignItems = 'center';
            avatar.style.justifyContent = 'center';
            avatar.style.fontSize = '0.75rem';
            
            if(role === 'user') {
                avatar.style.background = 'var(--bg-hover)';
                avatar.style.color = 'var(--text-muted)';
                avatar.innerHTML = '<i class="fas fa-user"></i>';
            } else {
                avatar.style.background = 'var(--bg-element)';
                avatar.style.color = 'var(--accent)';
                avatar.style.border = '1px solid var(--border)';
                avatar.innerHTML = '<i class="fas fa-robot"></i>';
            }

            // Bubble
            const bubble = document.createElement('div');
            bubble.className = `bubble ${role === 'user' ? 'bubble-user' : 'bubble-ai'} prose`;
            
            // Parse Markdown with Target Blank (Sanitized)
            const cleanHTML = DOMPurify.sanitize(marked.parse(text), { ADD_ATTR: ['target'] });
            bubble.innerHTML = cleanHTML;

            row.appendChild(avatar);
            row.appendChild(bubble);
            chatList.appendChild(row);
        }
